void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  runApp(const AttendanceCompleteApp());
}

// ======================== 1. النماذج (Models) ========================

enum AttendanceStatus {
  present('حاضر', Color(0xFF10B981), Color(0xFFECFDF5)),
  absent('غائب', Color(0xFFEF4444), Color(0xFFFEF2F2)),
  sickLeave('مرضي', Color(0xFFF59E0B), Color(0xFFFFFBEB)),
  regularLeave('عادية', Color(0xFF3B82F6), Color(0xFFEFF6FF));

  final String label;
  final Color color;
  final Color bgColor;
  const AttendanceStatus(this.label, this.color, this.bgColor);
}

class Employee {
  final String id;
  final String name;
  final String code;
  final String department;

  Employee({
    required this.id,
    required this.name,
    required this.code,
    required this.department,
  });

  Map<String, dynamic> toMap() => {
        'id': id,
        'name': name,
        'code': code,
        'department': department,
      };

  factory Employee.fromMap(Map<String, dynamic> map) => Employee(
        id: map['id'] ?? '',
        name: map['name'] ?? '',
        code: map['code'] ?? '',
        department: map['department'] ?? 'عام',
      );
}

// ======================== 2. التطبيق الرئيسي ========================

class AttendanceCompleteApp extends StatelessWidget {
  const AttendanceCompleteApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'سجل الحضور والغياب',
      debugShowCheckedModeBanner: false,
      locale: const Locale('ar', 'EG'),
      supportedLocales: const [Locale('ar', 'EG')],
      localizationsDelegates: const [
        GlobalMaterialLocalizations.delegate,
        GlobalWidgetsLocalizations.delegate,
        GlobalCupertinoLocalizations.delegate,
      ],
      theme: ThemeData(
        useMaterial3: true,
        fontFamily: 'sans-serif',
        colorScheme: ColorScheme.fromSeed(
          seedColor: const Color(0xFF1E3A8A),
          primary: const Color(0xFF1E3A8A),
          background: const Color(0xFFF8FAFC),
        ),
      ),
      home: const AttendanceHomePage(),
    );
  }
}

// ======================== 3. الشاشة الرئيسية مع إدارة الحالة والتخزين ========================

class AttendanceHomePage extends StatefulWidget {
  const AttendanceHomePage({super.key});

  @override
  State<AttendanceHomePage> createState() => _AttendanceHomePageState();
}

class _AttendanceHomePageState extends State<AttendanceHomePage> {
  DateTime _selectedDate = DateTime.now();
  List<Employee> _employees = [];
  Map<String, AttendanceStatus> _attendanceRecords = {};
  
  String _searchQuery = '';
  String? _selectedDepartment;

  @override
  void initState() {
    super.initState();
    _loadStoredData();
  }

  String _dateKey(DateTime date) => DateFormat('yyyy-MM-dd').format(date);

  Future<void> _loadStoredData() async {
    final prefs = await SharedPreferences.getInstance();
    
    // تحميل الموظفين
    final empString = prefs.getString('saved_employees');
    if (empString != null && empString.isNotEmpty) {
      final List decoded = jsonDecode(empString);
      _employees = decoded.map((e) => Employee.fromMap(e)).toList();
    } else {
      // موظفون تجريبيون عند أول فتح للتطبيق
      _employees = [
        Employee(id: '1', name: 'أحمد محمود', code: '101', department: 'تكنولوجيا المعلومات'),
        Employee(id: '2', name: 'سارة علي', code: '102', department: 'الموارد البشرية'),
        Employee(id: '3', name: 'محمد حسن', code: '103', department: 'الحسابات'),
        Employee(id: '4', name: 'فاطمة إبراهيم', code: '104', department: 'التسويق'),
      ];
      _saveEmployees();
    }

    // تحميل الحضور والغياب
    final attString = prefs.getString('saved_attendance');
    if (attString != null && attString.isNotEmpty) {
      final Map<String, dynamic> decoded = jsonDecode(attString);
      _attendanceRecords = decoded.map((key, value) =>
          MapEntry(key, AttendanceStatus.values.byName(value)));
    }

    setState(() {});
  }

  Future<void> _saveEmployees() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('saved_employees', jsonEncode(_employees.map((e) => e.toMap()).toList()));
  }

  Future<void> _saveAttendance() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
      'saved_attendance',
      jsonEncode(_attendanceRecords.map((key, val) => MapEntry(key, val.name))),
    );
  }

  void _toggleStatus(String empId, AttendanceStatus status) {
    final key = '${_dateKey(_selectedDate)}_$empId';
    setState(() {
      if (_attendanceRecords[key] == status) {
        _attendanceRecords.remove(key); // إلغاء الحالة عند النقر مرة ثانية
      } else {
        _attendanceRecords[key] = status;
      }
    });
    _saveAttendance();
  }

  void _markAllPresent() {
    final dateStr = _dateKey(_selectedDate);
    setState(() {
      for (var emp in _employees) {
        _attendanceRecords['${dateStr}_${emp.id}'] = AttendanceStatus.present;
      }
    });
    _saveAttendance();
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('تم تسجيل حضور جميع الموظفين لليوم المختار')),
    );
  }

  void _showAddOrEditDialog({Employee? employee}) {
    final nameCtrl = TextEditingController(text: employee?.name ?? '');
    final codeCtrl = TextEditingController(text: employee?.code ?? '');
    final deptCtrl = TextEditingController(text: employee?.department ?? '');

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(employee == null ? 'إضافة موظف جديد' : 'تعديل بيانات الموظف'),
        content: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              TextField(
                controller: nameCtrl,
                decoration: const InputDecoration(labelText: 'اسم الموظف *', border: OutlineInputBorder()),
              ),
              const SizedBox(height: 12),
              TextField(
                controller: codeCtrl,
                decoration: const InputDecoration(labelText: 'كود الموظف *', border: OutlineInputBorder()),
              ),
              const SizedBox(height: 12),
              TextField(
                controller: deptCtrl,
                decoration: const InputDecoration(labelText: 'القسم / الإدارة', border: OutlineInputBorder()),
              ),
            ],
          ),
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('إلغاء')),
          ElevatedButton(
            onPressed: () {
              final name = nameCtrl.text.trim();
              final code = codeCtrl.text.trim();
              final dept = deptCtrl.text.trim().isEmpty ? 'عام' : deptCtrl.text.trim();

              if (name.isEmpty || code.isEmpty) {
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(content: Text('يرجى ملء الاسم والكود')),
                );
                return;
              }

              setState(() {
                if (employee == null) {
                  _employees.add(Employee(
                    id: DateTime.now().millisecondsSinceEpoch.toString(),
                    name: name,
                    code: code,
                    department: dept,
                  ));
                } else {
                  final idx = _employees.indexWhere((e) => e.id == employee.id);
                  if (idx != -1) {
                    _employees[idx] = Employee(id: employee.id, name: name, code: code, department: dept);
                  }
                }
              });
              _saveEmployees();
              Navigator.pop(ctx);
            },
            child: const Text('حفظ'),
          ),
        ],
      ),
    );
  }

  void _deleteEmployee(Employee emp) {
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('تأكيد الحذف'),
        content: Text('هل تريد حذف الموظف ${emp.name} نهائياً؟'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('إلغاء')),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.red, foregroundColor: Colors.white),
            onPressed: () {
              setState(() => _employees.removeWhere((e) => e.id == emp.id));
              _saveEmployees();
              Navigator.pop(ctx);
            },
            child: const Text('حذف'),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final dateStr = _dateKey(_selectedDate);
    final arabicDate = DateFormat('EEEE، d MMMM yyyy', 'ar').format(_selectedDate);

    // حساب الإحصائيات
    int present = 0, absent = 0, sick = 0, regular = 0;
    for (var emp in _employees) {
      final st = _attendanceRecords['${dateStr}_${emp.id}'];
      if (st == AttendanceStatus.present) present++;
      if (st == AttendanceStatus.absent) absent++;
      if (st == AttendanceStatus.sickLeave) sick++;
      if (st == AttendanceStatus.regularLeave) regular++;
    }

    final filteredList = _employees.where((emp) {
      final matchQ = _searchQuery.isEmpty ||
          emp.name.contains(_searchQuery) ||
          emp.code.contains(_searchQuery) ||
          emp.department.contains(_searchQuery);
      final matchD = _selectedDepartment == null || emp.department == _selectedDepartment;
      return matchQ && matchD;
    }).toList();

    final allDepts = _employees.map((e) => e.department).toSet().toList();

    return Scaffold(
      appBar: AppBar(
        title: const Text('سجل الحضور والغياب (أوفلاين)', style: TextStyle(fontWeight: FontWeight.bold)),
        centerTitle: true,
        backgroundColor: const Color(0xFF1E3A8A),
        foregroundColor: Colors.white,
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () => _showAddOrEditDialog(),
        icon: const Icon(Icons.person_add),
        label: const Text('إضافة موظف جديد', style: TextStyle(fontWeight: FontWeight.bold)),
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(16, 16, 16, 90),
        children: [
          // شريط اختيار التاريخ
          Card(
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
            elevation: 2,
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Column(
                children: [
                  Row(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    children: [
                      IconButton(
                        icon: const Icon(Icons.arrow_forward_ios, size: 18),
                        onPressed: () => setState(() => _selectedDate = _selectedDate.subtract(const Duration(days: 1))),
                      ),
                      InkWell(
                        onTap: () async {
                          final picked = await showDatePicker(
                            context: context,
                            initialDate: _selectedDate,
                            firstDate: DateTime(2020),
                            lastDate: DateTime(2035),
                          );
                          if (picked != null) setState(() => _selectedDate = picked);
                        },
                        child: Row(
                          children: [
                            const Icon(Icons.calendar_month, color: Color(0xFF1E3A8A)),
                            const SizedBox(width: 8),
                            Text(arabicDate, style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16)),
                          ],
                        ),
                      ),
                      IconButton(
                        icon: const Icon(Icons.arrow_back_ios, size: 18),
                        onPressed: () => setState(() => _selectedDate = _selectedDate.add(const Duration(days: 1))),
                      ),
                    ],
                  ),
                  const Divider(),
                  Row(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    children: [
                      OutlinedButton.icon(
                        onPressed: () => setState(() => _selectedDate = DateTime.now()),
                        icon: const Icon(Icons.today, size: 16),
                        label: const Text('اليوم الحالي'),
                      ),
                      ElevatedButton.icon(
                        style: ElevatedButton.styleFrom(
                          backgroundColor: const Color(0xFF10B981),
                          foregroundColor: Colors.white,
                        ),
                        onPressed: _markAllPresent,
                        icon: const Icon(Icons.done_all, size: 18),
                        label: const Text('تسجيل الكل حضور'),
                      ),
                    ],
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 12),

          // شريط الإحصائيات
          SingleChildScrollView(
            scrollDirection: Axis.horizontal,
            child: Row(
              children: [
                _buildBadge('الإجمالي', '${_employees.length}', Colors.grey),
                _buildBadge('حضور', '$present', const Color(0xFF10B981)),
                _buildBadge('غياب', '$absent', const Color(0xFFEF4444)),
                _buildBadge('مرضي', '$sick', const Color(0xFFF59E0B)),
                _buildBadge('إجازة عادية', '$regular', const Color(0xFF3B82F6)),
              ],
            ),
          ),
          const SizedBox(height: 12),

          // شريط البحث
          TextField(
            onChanged: (v) => setState(() => _searchQuery = v.trim()),
            decoration: InputDecoration(
              hintText: 'ابحث بالاسم، الكود، أو القسم...',
              prefixIcon: const Icon(Icons.search),
              filled: true,
              fillColor: Colors.white,
              border: OutlineInputBorder(borderRadius: BorderRadius.circular(12), borderSide: BorderSide.none),
            ),
          ),
          const SizedBox(height: 8),

          // فلاتر الأقسام
          if (allDepts.isNotEmpty)
            SingleChildScrollView(
              scrollDirection: Axis.horizontal,
              child: Row(
                children: [
                  ChoiceChip(
                    label: const Text('كل الأقسام'),
                    selected: _selectedDepartment == null,
                    onSelected: (_) => setState(() => _selectedDepartment = null),
                  ),
                  const SizedBox(width: 6),
                  ...allDepts.map((d) => Padding(
                        padding: const EdgeInsets.only(right: 6),
                        child: ChoiceChip(
                          label: Text(d),
                          selected: _selectedDepartment == d,
                          onSelected: (_) => setState(() => _selectedDepartment = _selectedDepartment == d ? null : d),
                        ),
                      )),
                ],
              ),
            ),
          const SizedBox(height: 12),

          // بطاقات الموظفين
          if (filteredList.isEmpty)
            const Center(
              child: Padding(
                padding: EdgeInsets.all(40),
                child: Text('لا يوجد موظفون مضافون أو مطابقون للبحث'),
              ),
            )
          else
            ...filteredList.map((emp) {
              final status = _attendanceRecords['${dateStr}_${emp.id}'];
              return Card(
                elevation: 1.5,
                margin: const EdgeInsets.only(bottom: 10),
                shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
                child: Padding(
                  padding: const EdgeInsets.all(12),
                  child: Column(
                    children: [
                      Row(
                        children: [
                          Container(
                            padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
                            decoration: BoxDecoration(
                              color: const Color(0xFFDBEAFE),
                              borderRadius: BorderRadius.circular(8),
                            ),
                            child: Text(
                              emp.code,
                              style: const TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF1E3A8A)),
                            ),
                          ),
                          const SizedBox(width: 10),
                          Expanded(
                            child: Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: [
                                Text(emp.name, style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold)),
                                Text('القسم: ${emp.department}', style: TextStyle(color: Colors.grey[600], fontSize: 13)),
                              ],
                            ),
                          ),
                          PopupMenuButton(
                            itemBuilder: (ctx) => [
                              const PopupMenuItem(value: 'edit', child: Text('تعديل')),
                              const PopupMenuItem(value: 'delete', child: Text('حذف', style: TextStyle(color: Colors.red))),
                            ],
                            onSelected: (val) {
                              if (val == 'edit') _showAddOrEditDialog(employee: emp);
                              if (val == 'delete') _deleteEmployee(emp);
                            },
                          ),
                        ],
                      ),
                      const SizedBox(height: 10),
                      Row(
                        children: AttendanceStatus.values.map((s) {
                          final isSelected = status == s;
                          return Expanded(
                            child: Padding(
                              padding: const EdgeInsets.symmetric(horizontal: 2.5),
                              child: OutlinedButton(
                                style: OutlinedButton.styleFrom(
                                  padding: EdgeInsets.zero,
                                  backgroundColor: isSelected ? s.color : s.bgColor,
                                  foregroundColor: isSelected ? Colors.white : s.color,
                                  side: BorderSide(color: s.color, width: isSelected ? 2 : 1),
                                  shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
                                ),
                                onPressed: () => _toggleStatus(emp.id, s),
                                child: Text(
                                  s.label,
                                  style: TextStyle(
                                    fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,
                                    fontSize: 13,
                                  ),
                                ),
                              ),
                            ),
                          );
                        }).toList(),
                      ),
                    ],
                  ),
                ),
              );
            }),
        ],
      ),
    );
  }

  Widget _buildBadge(String title, String count, Color color) {
    return Container(
      margin: const EdgeInsets.only(left: 8),
      padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 6),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(10),
        border: Border.all(color: Colors.grey.shade300),
      ),
      child: Row(
        children: [
          Text(title, style: const TextStyle(fontSize: 13)),
          const SizedBox(width: 6),
          CircleAvatar(
            radius: 11,
            backgroundColor: color.withOpacity(0.15),
            child: Text(count, style: TextStyle(color: color, fontSize: 11, fontWeight: FontWeight.bold)),
          ),
        ],
      ),
    );
  }
}
